#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n; cin>>n;
    string s; cin>>s;
    int i=0, j=n-1;
    while(i<j){
        if(abs(s[i]-s[j])!=2 && s[i]!=s[j]){
            cout<<"NO\n"; return;
        }
        i++; j--;
    }
    cout<<"YES\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
