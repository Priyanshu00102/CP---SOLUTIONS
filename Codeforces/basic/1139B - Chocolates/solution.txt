#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n; cin>>n;
    int arr[n];
    for(int i=0; i<n; ++i){
        cin>>arr[i];
    }
    ll ans=arr[n-1]; int m=arr[n-1]-1;
    for(int i=n-2; i>=0; --i){
        if(m==0) break;
        int x=min(m,arr[i]);
        ans+=x;
        m=x-1;
        
    }
    cout<<ans<<"\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    // cin >> t;
    while (t--) solve();

    return 0;
}
