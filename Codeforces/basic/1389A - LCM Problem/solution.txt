#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int a,b; cin>>a>>b;
    if(b>=2*a){
        cout<<a<<" "<<2*a<<'\n';
        return;
    }
    cout<<-1<<" "<<-1<<'\n';

}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
