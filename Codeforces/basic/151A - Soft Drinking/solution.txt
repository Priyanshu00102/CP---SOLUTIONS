#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    ll n,k,l,c,d,p,nl,np;
    cin>>n >>k >>l >>c >>d >>p >>nl >>np;
    ll ml=(k*l)/nl;
    ll lime=c*d;
    ll salt=p/np;
    ll totol=min({ml,lime,salt});
    cout<<totol/n<<'\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    // cin >> t;
    while (t--) solve();

    return 0;
}
