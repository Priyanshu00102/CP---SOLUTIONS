#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n; cin>>n;
    ll sum=0;
    for(int i=0; i<n-1; ++i){
        int x;cin>>x;
        sum+=x;
    }
    cout<<-(sum)<<'\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
