// ---------- Solve ----------
void solve() {
   int n; cin>>n;
   string s; cin>>s;
   int cnt=0; int ans=0;
   for(int i=0; i<n; ++i){
       if(s[i]=='.') {
           cnt++;
           ans=max(ans,cnt);
       }
       else if(s[i]=='#')cnt=0;
   }
   int c=0;
   if(ans>=3) cout<<2<<'\n';
   else{
       for(int i=0; i<n; ++i){
           if(s[i]=='.') c++;
       }
       cout<<c<<'\n';
   }
}

    int t = 1;
    cin >> t;           // Comment this line if there is only one test case

    while (t--) {
        solve();
    }
