#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

void solve() {
    int n;cin>>n;
    vector<int>v(n);
    for(int &i:v){
        cin>>i;
    }
if(n<2) cout<<1<<'\n';
else{
    vector<int>ans(n,2);
    for(int i:ans){
        cout<<i<<" ";
    }
    cout<<'\n';
}
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
