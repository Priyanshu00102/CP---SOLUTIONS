#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n; cin>>n;
    vector<int>v(n);
    for(int i=0; i<n; ++i){
        int x; cin>>x;
        v[i]=x;
    }
    int cnt=0;
    for(int i=1; i<n; ++i){
        if(gcd(v[i-1],v[i])==abs(v[i-1]-v[i])){
            cnt++;
        }
    }
    cout<<cnt<<'\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
