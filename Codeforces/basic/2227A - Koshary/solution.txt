#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int a,n; cin>>a >>n;
    if(a%2!=0 && n%2!=0) cout<<"NO\n";
    else cout<<"YES\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
