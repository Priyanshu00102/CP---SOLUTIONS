#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    ll n,x,y,z; 
    cin>>n >>x >>y >>z;
    ll ans1=(n+x+y-1)/(x+y);
   ll ans2=0;
    if (z * x >= n)
    ans2 = (n + x - 1) / x;
else
    ans2 = z + (n - z*x + x + 10*y - 1) / (x + 10*y);
    cout<<min(ans1,ans2)<<'\n';

}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
