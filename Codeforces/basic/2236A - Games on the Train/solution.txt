#include <bits/stdc++.h>
using namespace std;

using ll = long long;

void solve() {
    int n;
    cin >> n;

    vector<int> v(n);
    for (int &i : v)
        cin >> i;

    int maxi = 0;
    int mini = INT_MAX;

    for (int i : v) {
        maxi = max(maxi, i);
        mini = min(mini, i);
    }

    ll ans = (maxi - mini) + 1;
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
