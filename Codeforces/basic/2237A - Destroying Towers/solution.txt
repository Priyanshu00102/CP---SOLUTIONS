#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n; cin>>n;
    int ans=INT_MAX; int sum=0;
    for(int i=0; i<n; ++i){
        int x; cin>>x;
        ans=min(ans,x);
        sum+=ans;
    }
    cout<<sum<<'\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
