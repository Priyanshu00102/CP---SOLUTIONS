#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n,m,k;
    cin>>n >>k >>m;
    if(k>m) cout<<"NO"<<'\n';
    else {
        cout<<"YES"<<'\n';
        int sum=0;
        for(int i=1; i<=k-1; ++i){
            cout<<1<<" ";
            sum++;
        }
        cout<<m-sum<<" ";
        for(int i=k+1; i<=n; ++i){
            cout<<1<<" ";
        }
    }
    cout<<'\n';

}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
