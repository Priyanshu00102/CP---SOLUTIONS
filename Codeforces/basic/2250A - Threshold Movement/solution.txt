#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        int n;
        cin >> n;

        vector<int> w(n);

        int low = 0;
        int high = 1000000001;

        for (int i = 0; i < n; i++) {
            cin >> w[i];

            if (i % 2 == 0) {      // positions 1,3,5,... -> Right
                high = min(high, w[i]);   // k < w[i]
            } else {               // positions 2,4,6,... -> Left
                low = max(low, w[i]);     // k > w[i]
            }
        }

        if (n % 2 == 1) {
            cout << "NO\n";
        } else if (low + 1 < high) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
}
