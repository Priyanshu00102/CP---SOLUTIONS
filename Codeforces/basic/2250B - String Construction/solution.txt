#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
   int n,k; cin>>n>>k;
   if(n-2<k) {
    cout<<-1<<endl;
    return;
   }
   int zero=k/2;
   int one=k-zero;
   for(int i=1; i<=zero+1; i++){
    cout<<0;
    n--;
   }
    for(int i=1; i<=one+1; i++){
    cout<<1;
    n--;
   }
    for(int i=1; i<=n; i++){
    if(i%2==0){
        cout<<1;
    }else {
        cout<<0;
    }
   }
   cout<<endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
