#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int a, b, c;
    cin>>a >>b >>c;
    if(a==b||b==c||c==a) {
        cout<<0<<'\n';
        return;
    }
    vector<int>v={a,b,c};
    sort(v.begin(), v.end());
    cout<<min(v[1]-v[0],v[2]-v[1])<<'\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
