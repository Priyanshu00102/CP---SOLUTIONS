#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int x,y; cin>>x>>y;
    int ans=0;
    while(x>0&&y>0){
        if(x<y) swap(x,y);
        ans+=x/y;
        x=x%y;
    }
    cout<<ans<<'\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
