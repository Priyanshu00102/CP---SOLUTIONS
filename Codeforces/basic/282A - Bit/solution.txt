#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n; cin>>n;
    
    int ans=0;
    while(n--){
        string s; cin>>s;
    if(s=="X++"||s=="++X") ans+=1;
    else ans-=1; 
    }
    cout<<ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    // cin >> t;
    while (t--) solve();

    return 0;
}
