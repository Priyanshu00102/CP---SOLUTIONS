#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n, k;
    cin >> n >> k;

    vector<pair<int,int>> v(n);

    for(int i = 0; i < n; i++){
        cin >> v[i].first;
        v[i].second = i + 1;
    }

    sort(v.begin(), v.end());

    int sum = 0;
    vector<int> ans;

    for(int i = 0; i < n; i++){
        if(sum + v[i].first <= k){
            sum += v[i].first;
            ans.push_back(v[i].second);
        }
        else {
            break;
        }
    }

    cout << ans.size() << "\n";

    for(int i : ans){
        cout << i << " ";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    solve();

    return 0;
}
