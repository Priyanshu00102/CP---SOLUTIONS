#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int a,b,c;
    cin>>a >>b >>c;
    int maxi=max({a,b,c});
    int mini=min({a,b,c});
    cout<<maxi-mini<<'\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    // cin >> t;
    while (t--) solve();

    return 0;
}
