#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    ll a,b,c; cin>>a >>b >>c;
    ll maxi=max({a,b,c});
    if(a+b<maxi){
        maxi=a+b;
    }else if(b+c<maxi) maxi=b+c;
    else if(c+a<maxi) maxi=c+a;
    ll mini=min({a,b,c});
    cout<<maxi-mini<<'\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    cin >> t;
    while (t--) solve();

    return 0;
}
